ary_supreme_blood_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_tale.dds"
	}
	decision_group_type = ary_traditions_group_type
	desc = ary_supreme_blood_decision_desc
	selection_tooltip = ary_supreme_blood_decision_tooltip
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		OR = {
			culture = { has_cultural_tradition = tradition_supremeblood }
			culture = { has_cultural_tradition = tradition_supremeblood2 }
			culture = { has_cultural_tradition = vv_tradition_supremeblood }
			culture = { has_cultural_tradition = vv_tradition_supremeblood2 }
		}
	}
	
	is_valid = {
		OR = {
			prestige_level >= 3
			piety_level >= 3
		}
		OR = {
			age >= 100
			has_trait = incapable
			has_trait = leper
			has_trait = maimed
			has_trait = gout_ridden
			has_trait = lovers_pox
			has_trait = infertile
			has_trait = depressed_1
			has_trait = lunatic_1
			has_trait = infirm
		}
	}

	effect = {
		hidden_effect = {
			add_piety_level = 3
			add_prestige_level = 3
			root.dynasty = {
				add_dynasty_prestige >= 1000
			}
		}
		death = { death_reason = death_peaceful }
		show_as_tooltip = {
			add_piety_level = 3
			add_prestige_level = 3
			root.dynasty = {
				add_dynasty_prestige >= 1000
			}
		}
		player_heir = {
			add_character_modifier = {
				modifier = ary_true_successor_modifier
				years = 15
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}
