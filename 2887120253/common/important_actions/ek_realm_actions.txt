action_can_create_mystical_birthright_head_of_faith_title = {
	check_create_action = {
		if = {
			limit = {
				faith = {
					NOT = { exists = religious_head }
					has_doctrine = doctrine_mystical_birthright_head
					can_create_mystical_birthright_head_of_faith_title_trigger = yes
				}
			}
			try_create_important_action = {
				important_action_type = action_can_create_mystical_birthright_head_of_faith_title
				actor = root
			}
		}
	}

	unimportant = {
		short_term_gold < 300
	}

	effect = {
		scope:actor.faith = {
			open_view_data = {
				view = faith
			}
		}
	}
}

action_fate_of_elsweyr = {
	type = alert
	is_dangerous = yes
	combine_into_one = no
	icon = action_mandala_red
	check_create_action = {
		if = {
			limit = {
				NOT = { #Make sure dissolution didn't happen yet
					is_target_in_global_variable_list = {
						name = elsweyr_variables
						target = flag:elsweyr_dissolved
					} 
				}
				has_title = title:e_elsweyr # gotta hold this for it to be very relevant
				NOT = { current_date >= 3410.1.1 } # stop showing this at some point
			}
			try_create_important_action = {
				important_action_type = action_fate_of_elsweyr
				actor = root
			}
		}
	}
}
