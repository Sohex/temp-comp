	skill = diplomacy
	group = culture_group_regional
	culture_era = culture_era_tribal
	icon = @pariah_01
	
	potential = {
		trigger_if = {
			limit = { years_from_game_start < 1 }
			OR = {
				has_cultural_pillar = heritage_goblinken
				has_cultural_pillar = heritage_orcish
				this = culture:falmer_betrayed
				any_parent_culture_or_above = { this = culture:falmer_betrayed }
				has_cultural_pillar = heritage_atmoran_giant
			}
		}
		trigger_else = {
			always = no
		}
	}

	character_modifier = {
		same_culture_opinion = 10
	}
	
	custom = innovation_pariah_folk_cultural_acceptance
	custom = innovation_pariah_folk_marriage
	custom = innovation_pariah_folk_revocation
	custom = innovation_pariah_folk_persecution
	custom = innovation_pariah_folk_can_be_removed

	flag = global_regional
	flag = tribal_era_regional
}
