ek_uses_special_succession_law = { # Something in your faith/culture makes you use a special succ law even if you have a feudal-ish government
	OR = {
		faith = { has_doctrine_parameter = rule_by_might_succession_active } # Rule by Might
	}
}

can_have_dynastic_acclamation_succession_law_trigger = {
	top_liege = this
	government_allows = administrative
	OR = {
		has_cyrodilic_empire_trigger = yes
		AND = {
			culture = { has_cultural_tradition = tradition_imperial_legacy }
			highest_held_title_tier >= tier_kingdom
		}
	}
}

can_keep_dynastic_acclamation_succession_law_trigger = {
	trigger_if = {
		limit = {
			has_realm_law = dynastic_acclamation_succession_law 
		}
		can_have_dynastic_acclamation_succession_law_trigger = yes
	}
}
