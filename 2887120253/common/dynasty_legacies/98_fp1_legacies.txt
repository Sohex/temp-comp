### EK EDIT: Pillage unlocked by Practiced Pirates or pirate government
fp1_pillage_legacy_track = {
	is_shown = {
		has_dlc_feature = the_northern_lords
		OR = {
			has_game_rule = unrestricted_dynasty_legacies_all
			AND = {
				has_game_rule = unrestricted_dynasty_legacies_player_only
				is_ai = no
			}
			dynasty = {
				OR = {
					has_dynasty_perk = fp1_pillage_legacy_1
					dynast = { has_government = pirate_government }
					dynast.culture = { has_cultural_parameter = unlock_pillage_legacy }
				}
			}
		}
	}
}
