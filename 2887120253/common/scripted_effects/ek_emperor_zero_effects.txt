
#Spawn Emperor Zero
spawn_emperor_zero_character_effect = {
    #Pick Imperial king who's good at martial. Colovians are prefered
	if = {
		limit = { 
			exists = title:e_colovian_estates.holder
			title:e_colovian_estates.holder.culture = { has_cultural_pillar = heritage_cyrodiilic }
		}
		title:e_colovian_estates.holder = {  save_scope_as = emperor_zero_character }
	}
	if = {
        limit = {
            NOT = { exists = scope:emperor_zero_character }
        }
		random_ruler = {
			limit = {
				culture = { has_cultural_tradition = tradition_colovian_core }
				highest_held_title_tier >= tier_kingdom
				OR = {
					government_has_flag = government_is_autocracy
					government_has_flag = government_is_administrative
				}
				has_trait = education_martial
				calc_true_if = {
					amount >= 2				
					has_trait = ambitious
					has_trait = brave
					has_trait = zealous
				}
			}
			save_scope_as = emperor_zero_character
		}
	}

    #If there is no king that fits, then spawn Culhecain in Falkreath
    if = {
        limit = {
            NOT = { exists = scope:emperor_zero_character }
        }
	    title:c_falkreath.title_province = {
	    	save_scope_as = cuhlecain_birthplace
	    }
        create_character = {
            name = "Cuhlecain" # Cuhlecain
            location = scope:cuhlecain_birthplace
            template = cuhlecain_character_template
            save_scope_as = emperor_zero_character
        }
		#EK NOTE: useless for now, only relevant in 812 bookmark
		# set_global_variable = {
		# 	name =  cuhlecain_was_born
		# 	value = scope:emperor_zero_character
		# }
		scope:emperor_zero_character = {
			add_character_flag = flag_spawned_cuhlecain
			house = { set_house_name = dynn_Curtis }
			dynasty = { set_dynasty_name = dynn_Curtis }
		}
    }

    #Buff emperor zero
    scope:emperor_zero_character = {
		#save_scope_as = story_owner
		create_story = story_emperor_zero
        # Make temporarily immune to disease
        add_character_flag = {
            flag = immune_to_disease
            years = 15
        }
		add_character_flag = {
			flag = immune_to_harm_events
		}
		add_prestige = 15000
		add_gold = 5000
		form_the_emperor_zero_empire_effect = yes
		add_legitimacy = 5000 #After title formation
        #trigger_event = conqueror.0001
        if = { #Give Cuhlecain maa regiments
			limit = {
				has_character_flag = flag_spawned_cuhlecain
			}
			create_maa_regiment = {
				type = armored_legionaires
				check_can_recruit = no
				size = 10
			}
			create_maa_regiment = {
				type = armored_legionaires
				check_can_recruit = no
				size = 10
			}
			create_maa_regiment = {
				type = equestrian_brigade
				check_can_recruit = no
				size = 10
			}
			create_maa_regiment = {
				type = trebuchet
				check_can_recruit = no
				size = 10
			}
		}
        if = {
            limit = {
                has_trait = education_martial
                NOT = { has_perk = peacemaker_perk }
            }
            add_perk = peacemaker_perk
        }
        add_trait = flexible_leader
        add_trait = emperor_zero
    }

	if = {
		limit = {
			scope:emperor_zero_character = {
				is_ai = yes 
			}
		}
        #Normal event
        every_player = {
            limit = {
                NOT = {
                    this = scope:emperor_zero_character
                }
            }
            trigger_event = emperor_zero.1002 # Emperor Zero spawned notification
        }
	}
	else = { #If player is Emperor Zero
		scope:emperor_zero_character = {
			trigger_event = emperor_zero.1003
		}
	}
}

# Buff Emperor Zero, give land to Cuhlecain
form_the_emperor_zero_empire_effect = {
	save_scope_as = emperor_zero_founder

    if = {
        limit = {
            is_landed = no
        }
		if = {
			limit = { #If Falkreath is ruled by player only give him a county inside Falkreath
				title:k_falkreath.holder ?= { is_ai = no }
			}
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = title_change
				add_claim_on_loss = no
			}
			title:c_shriekwind = {
				change_title_holder = {
					holder = scope:emperor_zero_founder
					change = scope:title_change
				}
			}
			resolve_title_and_vassal_change = scope:title_change
		}
		else = {
			#Create the title
			title:k_falkreath = { add_to_list = starting_titles_list }
			title:d_falkreath = { add_to_list = starting_titles_list }
			title:d_falkreath = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_county
						holder = { is_ai = yes }
					}
					add_to_list = starting_titles_list
				}
			}
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = title_change
				add_claim_on_loss = no
			}
			every_in_list = {
				list = starting_titles_list
				change_title_holder = {
					holder = scope:emperor_zero_founder
					change = scope:title_change
				}
			}
			resolve_title_and_vassal_change = scope:title_change
		}
    }

	if = {
		limit = {
			is_independent_ruler = no
		}
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = indpendence_change
			add_claim_on_loss = no
		}
		becomes_independent = {
			change = scope:indpendence_change
		}
		resolve_title_and_vassal_change = scope:indpendence_change
	}

	if = {
		limit = {
			capital_county = {
				title_province = {
					has_holding_type = tribal_holding
				}
			}
		}
		capital_county = {
			title_province = {
				set_holding_type = castle_holding
			}
		}
	}
	
    if = {
		limit = {
			government_has_flag = government_is_autocracy
		}
		if = {
			limit = { has_realm_law = crown_authority_0 }
			remove_realm_law = crown_authority_0
		}
		if = {
			limit = { has_realm_law = crown_authority_1 }
			remove_realm_law = crown_authority_1
		}
		if = {
			limit = { has_realm_law = crown_authority_3 }
			remove_realm_law = crown_authority_3
		}
		add_realm_law_skip_effects = crown_authority_2
	
		add_realm_law_skip_effects = single_heir_succession_law
	}

	spawn_emperor_zero_troops_effect = yes

	#Make sure that Emperor Zero is crowned
	#Anointed if he must be anointed
	if = {
		limit = {
			OR = {
				has_realm_law = uncrowned
				has_realm_law = crowned_king
			}
			faith = { has_doctrine = doctrine_imperial_anointment }
		}
		if = {
			limit = {
				NOT = {
					has_game_rule = coronation_laws_off
				}
			}
			add_realm_law = crowned_emperor
		}
		set_variable = crowned_emperor_var
		if = {
			limit = {
				has_variable = crowned_king_var
			}
			remove_variable = crowned_king_var
		}
	}
	else_if = {
		limit = {
			has_realm_law = uncrowned
		}
		if = {
			limit = {
				NOT = {
					has_game_rule = coronation_laws_off
				}
			}
			add_realm_law = crowned_king
		}
		set_variable = crowned_king_var
	}
}

emperor_zero_war_target_evaluation_and_declaration_effect = {
	save_scope_as = actor
	# Set the Emperor Zero as reference point
	save_scope_as = emperor_zero
	add_character_flag = {
		flag = free_emperor_zero_cb
		days = 14
	}
	# Select a new target
	random_neighboring_top_liege_realm_owner = {
		limit = {
			NOT = { is_at_war_with = scope:emperor_zero }
			NOT = { is_allied_to = scope:emperor_zero }
			this != scope:emperor_zero
			top_liege != scope:emperor_zero
			save_temporary_scope_as = truce_check
			NOT = {
				scope:emperor_zero = {
					any_truce_target = {
						this = scope:truce_check
					}
				}
			}
			save_temporary_scope_as = recipient
			emperor_zero_subjugation_shown_recipient_trigger = yes
			emperor_zero_subjugation_valid_recipient_trigger = yes
		}
		weight = {
			base = 0
			modifier = {
				add = emperor_zero_invasion_target_character_weight
				always = yes
			}
		}
		save_temporary_scope_as = next_invasion_target
		primary_title = {
			save_temporary_scope_as = next_invasion_title_target
		}
	}
	if = {
		limit = {
			exists = scope:next_invasion_target
		}
		if = {
			limit = { # Declare war on every same-tier count or duke in same kingdom
				scope:next_invasion_target = {
					OR = {
						highest_held_title_tier = tier_county
						highest_held_title_tier = tier_duchy
					}
				}
			}
			scope:next_invasion_title_target = {
				add_to_temporary_list = all_titles_to_declare_on
				kingdom = {
					every_in_de_jure_hierarchy = { # Find all other same-tier titles in same de jure kingdom
						continue = {
							tier > tier_county
							#EK NOTE: vanilla uses this for conqueror story cycles, but this doesn't work that great for us
							# Stop searching if you hit the Emperor's territory at any point
							# trigger_if = {
							# 	limit = { exists = holder }
							# 	holder != scope:emperor_zero
							# 	holder.top_liege != scope:emperor_zero
							# }
						}
						limit = {
							exists = this
							OR = {
								tier = tier_county
								tier = tier_duchy
							}
							holder ?= {
								top_liege = this
								this != scope:emperor_zero
								top_liege != scope:emperor_zero
								NOT = { is_at_war_with = scope:emperor_zero }
								NOT = { is_allied_to = scope:emperor_zero }
								NOT = { is_at_war_with = scope:emperor_zero }
								save_temporary_scope_as = truce_check
								NOT = {
									scope:emperor_zero = {
										any_truce_target = {
											this = scope:truce_check
										}
									}
								}
								save_temporary_scope_as = recipient
								emperor_zero_subjugation_shown_recipient_trigger = yes
								emperor_zero_subjugation_valid_recipient_trigger = yes
								in_diplomatic_range = scope:emperor_zero
							}
						}
						add_to_temporary_list = all_titles_to_declare_on
					}
				}
				every_in_list = {
					list = all_titles_to_declare_on
					limit = {
						holder = {
							# An extra check to make sure we haven't snuck any Mongol territory in
							NOT = { this = scope:emperor_zero }
							NOT = { top_liege = scope:emperor_zero }
	
							# Standard conditions
							NOT = { is_at_war_with = scope:emperor_zero }
							save_temporary_scope_as = truce_check

							save_temporary_scope_as = recipient
							emperor_zero_subjugation_shown_recipient_trigger = yes
							emperor_zero_subjugation_valid_recipient_trigger = yes
							in_diplomatic_range = scope:emperor_zero
						}
						NOT = {
							scope:emperor_zero = {
								any_truce_target = {
									this = scope:truce_check
								}
							}
						}
					}
					holder = {
						add_to_list = title_holders_declared_on
					}
				}
				every_in_list = {
					list = title_holders_declared_on
					save_scope_as = recipient
					scope:emperor_zero = {
						run_interaction = {
							interaction = emperor_zero_demand_subjugation_interaction
							actor = scope:emperor_zero
							recipient = scope:recipient
							send_threshold = decline
						}
					}
				}
			}
		}	
		else = {
			scope:next_invasion_target = {
				save_scope_as = recipient
			}
			scope:emperor_zero = {
				run_interaction = {
					interaction = emperor_zero_demand_subjugation_interaction
					actor = scope:emperor_zero
					recipient = scope:recipient
					send_threshold = decline
				}
			}
		}
	}
}

emperor_zero_accept_subjugation_effect = {
	#If kingdom or above... loses titles
	if = {
		limit = {
			scope:recipient = {
				highest_held_title_tier >= tier_kingdom
			}
		}
		scope:recipient = {
			hidden_effect = {
				every_vassal = {
					limit = {
						highest_held_title_tier = tier_duchy
					}
					create_title_and_vassal_change = {
						type = swear_fealty
						save_scope_as = change
					}
					change_liege = {
						liege = scope:actor
						change = scope:change
					}
					add_opinion = {
						modifier = became_vassal
						target = scope:actor
						opinion = 10
					}
					resolve_title_and_vassal_change = scope:change
				}
			}
			#Have to grant them a duchy if they don't have any...
			if = {
				limit = {
					NOT = {
						any_held_title = {
							tier = tier_duchy
						}
					}
				}
				if = {
					limit = {
						capital_county.duchy ?= {
							NOT = {
								exists = holder
							}
						}
					}
					create_title_and_vassal_change = {
						type = created
						save_scope_as = change
						add_claim_on_loss = no
					}
					capital_county.duchy = {
						change_title_holder = {
							holder = scope:recipient
							change = scope:change
						}
					}
					resolve_title_and_vassal_change = scope:change
				}
			}
			every_held_title = {
				limit = {
					tier >= tier_kingdom
				}
				scope:recipient = {
					destroy_title = prev
				}
			}
		}
	}
	
	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = change
	}
	scope:recipient = {
		change_liege = {
			liege = scope:actor
			change = scope:change
		}
		hidden_effect = {
			add_opinion = {
				modifier = became_vassal
				target = scope:actor
				opinion = 10
			}
		}
	}
	resolve_title_and_vassal_change = scope:change

	#If Emperor is a pluralist, they always get religious freedom
	scope:recipient = {
		if = {
			limit = {
				#Actor is still nomadic or pluralist and thus doesn't care about religion
				scope:actor = {
					faith = {
						has_doctrine = doctrine_pluralism_pluralistic
					}
				}
				OR = {
					government_has_flag = government_is_feudal
					government_has_flag = government_is_autocracy
				}
			}
			hidden_effect = {
				if = {
					limit = { is_independent_ruler = no } # To avoid errors during tooltip generation (prior to the contract being generated)
					vassal_contract_set_obligation_level = { type = religious_rights level = 1 }
				}
			}
			custom_tooltip = religious_protection_tt
		}
	}

	# Special terms for feudal contracts
	scope:recipient = {
		if = {
			limit = {
				government_has_flag = government_is_feudal
			}
			hidden_effect = {
				if = {
					limit = { is_independent_ruler = no } # To avoid errors during tooltip generation (prior to the contract being generated)
					vassal_contract_set_obligation_level = { type = feudal_government_taxes level = 2 }
					vassal_contract_set_obligation_level = { type = feudal_government_levies level = 2 }
				}
			}
			custom_tooltip = normal_obligations_taxes
			custom_tooltip = normal_obligations_levies
		}
	}

	#Get legions
	scope:actor = {
		scope:recipient = {
			save_scope_as = army_giver
		}
		custom_tooltip = gok_quality_army_gain_tt
		if = {
			limit = {
				exists = scope:recipient.primary_title
			}
			scope:recipient.primary_title = {
				save_scope_as = army_origin_realm
			}
		}
		else = {
			scope:recipient.location.county.kingdom = {
				save_scope_as = army_origin_realm
			}
		}
		hidden_effect = {
			random_list = {
				#Nord Huscarls
				100 = {
					trigger = {
						scope:army_giver.culture = { has_cultural_parameter = unlock_maa_huscarls }
					}

						name = gok_subjugated_army_name_subject_troops

						inheritable = yes

true_emperor_war_target_evaluation_and_declaration_effect = {
	save_scope_as = actor
	# Set the Emperor Zero as reference point
	save_scope_as = emperor_zero
	add_character_flag = {
		flag = free_emperor_zero_cb
		days = 14
	}
	# Select a new target
	random_neighboring_top_liege_realm_owner = {
		limit = {
			NOT = { is_at_war_with = scope:emperor_zero }
			NOT = { is_allied_to = scope:emperor_zero }
			this != scope:emperor_zero
			top_liege != scope:emperor_zero
			save_temporary_scope_as = truce_check
			NOT = {
				scope:emperor_zero = {
					any_truce_target = {
						this = scope:truce_check
					}
				}
			}
			save_temporary_scope_as = recipient
			emperor_zero_subjugation_shown_recipient_trigger = yes
			emperor_zero_subjugation_valid_recipient_trigger = yes
		}
		weight = {
			base = 0
			modifier = {
				add = emperor_zero_invasion_target_character_weight
				always = yes
			}
		}
		save_temporary_scope_as = next_invasion_target
		primary_title = {
			save_temporary_scope_as = next_invasion_title_target
		}
	}
	if = {
		limit = {
			exists = scope:next_invasion_target
		}
		if = {
			limit = { # Declare war on every same-tier count or duke in same kingdom
				scope:next_invasion_target = {
					OR = {
						highest_held_title_tier = tier_county
						highest_held_title_tier = tier_duchy
					}
				}
			}
			scope:next_invasion_title_target = {
				add_to_temporary_list = all_titles_to_declare_on
				kingdom = {
					every_in_de_jure_hierarchy = { # Find all other same-tier titles in same de jure kingdom
						continue = {
							tier > tier_county
							# Stop searching if you hit the Emperor's territory at any point
							trigger_if = {
								limit = { exists = holder }
								holder != scope:emperor_zero
								holder.top_liege != scope:emperor_zero
							}
						}
						limit = {
							exists = this
							OR = {
								tier = tier_county
								tier = tier_duchy
							}
							holder ?= {
								top_liege = this
								this != scope:emperor_zero
								top_liege != scope:emperor_zero
								NOT = { is_at_war_with = scope:emperor_zero }
								NOT = { is_allied_to = scope:emperor_zero }
								NOT = { is_at_war_with = scope:emperor_zero }
								save_temporary_scope_as = truce_check
								NOT = {
									scope:emperor_zero = {
										any_truce_target = {
											this = scope:truce_check
										}
									}
								}
								save_temporary_scope_as = recipient
								emperor_zero_subjugation_shown_recipient_trigger = yes
								emperor_zero_subjugation_valid_recipient_trigger = yes
								in_diplomatic_range = scope:emperor_zero
							}
						}
						add_to_temporary_list = all_titles_to_declare_on
					}
				}
				every_in_list = {
					list = all_titles_to_declare_on
					limit = {
						holder = {
							# An extra check to make sure we haven't snuck any Mongol territory in
							NOT = { this = scope:emperor_zero }
							NOT = { top_liege = scope:emperor_zero }
	
							# Standard conditions
							NOT = { is_at_war_with = scope:emperor_zero }
							save_temporary_scope_as = truce_check

							save_temporary_scope_as = recipient
							emperor_zero_subjugation_shown_recipient_trigger = yes
							emperor_zero_subjugation_valid_recipient_trigger = yes
							in_diplomatic_range = scope:emperor_zero
						}
						NOT = {
							scope:emperor_zero = {
								any_truce_target = {
									this = scope:truce_check
								}
							}
						}
					}
					holder = {
						add_to_list = title_holders_declared_on
					}
				}
				every_in_list = {
					list = title_holders_declared_on
					save_scope_as = recipient
					scope:emperor_zero = {
						run_interaction = {
							interaction = emperor_zero_demand_subjugation_interaction
							actor = scope:emperor_zero
							recipient = scope:recipient
							send_threshold = decline
						}
					}
				}
			}
		}	
		else = {
			scope:next_invasion_target = {
				save_scope_as = recipient
			}
			scope:emperor_zero = {
				run_interaction = {
					interaction = emperor_zero_demand_subjugation_interaction
					actor = scope:emperor_zero
					recipient = scope:recipient
					send_threshold = decline
				}
			}
		}
	}
