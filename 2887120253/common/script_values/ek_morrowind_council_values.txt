odds_house_war_scheme_misc_value = {
	add = {
		value = 0
		desc = scheme_odds.misc_modifiers
		
		add = {
			value = scope:owner.prestige_level
			multiply = 1
		}
		add = {
			value = scope:target.holder.prestige_level
			multiply = -1
		}
		add = {
			value = scope:owner.piety_level
			multiply = 1
		}
		add = {
			value = scope:target.holder.piety_level
			multiply = -1
		}
		if = {
			limit = {
				scope:target.holder.house = scope:owner.top_liege.house
			}
			add = -12
		}
		if = {
			limit = {
				scope:target.holder = { is_allowed_to_vote = yes }
			}
			add = -50
		}
		if = {
			limit = {
				scope:owner.house = {
					is_powerful_family = yes
					is_dominant_family = no
				}
			}
			add = 8
		}
		if = {
			limit = {
				scope:target.holder.house = {
					is_powerful_family = yes
					is_dominant_family = no
				}
			}
			add = -16
		}
	}
}
