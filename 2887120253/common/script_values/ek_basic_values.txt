# Weight used in selecting targets for the automated Imperial invasions
emperor_zero_invasion_target_character_weight = {
	value = 0.001
	every_sub_realm_county = {
		if = {
			limit = { #Colovia + Heartlands
				title_province = {
					geographical_region = emperor_zero_core_territory
				}
			}
			add = 300000
		}
		if = {
			limit = { #Nibenay
				title_province = {
					geographical_region = mundus_tamriel_cyrodiil_nibenay
				}
			}
			add = 150000
		}
		else_if = {
			limit = {
				title_province = { #Skyrim
					geographical_region = mundus_tamriel_skyrim
				}
			}
			add = 20000
		}
		else_if = {
			limit = {
				title_province = { #High Rock and Hammerfell
					OR = {
						geographical_region = mundus_tamriel_high_rock
						geographical_region = mundus_tamriel_hammerfell_no_alikr
					}
				}
			}
			add = 10000
		}
		else_if = {
			limit = {
				title_province = {
					OR = { #Elsweyr, Valenwood and parts of Black Marsh
						geographical_region = mundus_tamriel_elsweyr
						geographical_region = mundus_tamriel_valenwood
						geographical_region = mundus_tamriel_black_marsh_major_cities

					}
				}
			}
			add = 8000
		}
		else_if = {
			limit = {
				title_province = {
					OR = { #Morrowind, Alinor - elf provinces
						geographical_region = mundus_tamriel_morrowind_mainland
						geographical_region = mundus_tamriel_summerset_isles
					}
				}
			}
			add = 4000
		}
	}
	#divide = realm_size
}
