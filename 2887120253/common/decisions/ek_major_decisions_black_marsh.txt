#Restore Arnesia
recreate_lost_kingdom_arnesia_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/dunmer_council.dds" }

	desc = recreate_lost_kingdom_arnesia_decision_desc
	selection_tooltip = recreate_lost_kingdom_arnesia_decision_tooltip
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_shown = {
		is_ruler = yes
		is_landed = yes
		exists = dynasty # Everyone should have a dynasty, but in case they don't...

		capital_province = { geographical_region = mundus_tamriel_black_marsh_arnesia } #if you live within Arnesia

		culture = {
			OR = { #this is kinda a Cantemiric thing
				this = culture:cantemiric
				any_parent_culture_or_above = { this = culture:cantemiric }
			}
		}

		NOT = {	#You can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:recreate_lost_kingdom_arnesia_decision
			}
		}
		highest_held_title_tier <= tier_duchy
	}
	
	is_valid = {
		is_independent_ruler = yes
		
		prestige_level >= high_prestige_level
		has_title = title:d_arnesia
		completely_controls_region = mundus_tamriel_black_marsh_arnesia
		
		title:c_arnesia = { development_level >= 10 }
		has_feudal_or_admin_government = yes
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	cost = {
		gold = 500
	}

	effect = {
		show_as_tooltip = {
			recreate_arnesia_decision_scripted_effect = yes
		}
		save_scope_as = founder
		trigger_event = {
			id = south_tamriel.0001
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:recreate_lost_kingdom_arnesia_decision
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

### Form Cantemiris
form_cantemiris_empire_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/dunmer_council.dds" }

	desc = form_cantemiris_empire_decision_desc

	selection_tooltip = form_cantemiris_empire_decision_tooltip

	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 120
		empire = 0
		hegemony = 0
	}

	is_shown = {
		is_landed = yes
		highest_held_title_tier = tier_kingdom
		exists = capital_province
		unify_kingdom_government_trigger = yes
		capital_province = { geographical_region = mundus_tamriel_black_marsh_padomaic_marshes }
		
		culture = {
			OR = { #this is kinda a Cantemiric thing
				this = culture:cantemiric
				any_parent_culture_or_above = { this = culture:cantemiric }
			}
		}
		
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_created_cantemiris_empire
			}
		}
	}

	is_valid = {
		completely_controls_region = mundus_tamriel_black_marsh_padomaic_marshes #we check for padomaic_marshes, as it covers Thorn + Archon (and therefore also Arnesia)
		unify_kingdom_law_trigger = yes
		prestige_level >= 3
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	cost = {
		# Based on Revive Greater Armenia
		prestige = 2000
		piety = 1000
	}

	effect = {
		show_as_tooltip = {
			form_cantemiris_empire_scripted_effect = yes
		}

		if = {
			limit = { has_dlc_feature = legends }
			create_legend_seed = {
				type = legitimizing
				quality = illustrious
				chronicle = new_title
				properties = {
					title = title:e_cantemiris
					founder = root
				}
			}
		}
		
		trigger_event = {
			id = ektitulartitle.1301
		}
		#Notify other players.
		every_player = {
			limit = {
				NOT = { this = root }
				is_within_diplo_range = { CHARACTER = root }
			}
			trigger_event = ektitulartitle.1302
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_created_cantemiris_empire
			}
		}
	}

	confirm_text = form_cantemiris_empire_decision_confirm
	
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

create_ojel_culture_decision = {
	picture = { reference ="gfx/interface/illustrations/decisions/decision_destiny_goal.dds" }

	desc = create_ojel_culture_decision_desc
    selection_tooltip = create_ojel_culture_decision_tooltip
    decision_group_type = major


	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 120
		empire = 0
		hegemony = 0
	}

    is_shown = {
        is_landed = yes
		is_ruler = yes
		highest_held_title_tier <= tier_kingdom #You don't want a subservient culture if you're a high king
        culture = { has_cultural_tradition = tradition_agaceph_core } #Agaceph only
		capital_province = {
			OR = {
				geographical_region = mundus_tamriel_black_marsh_gideon
				geographical_region = mundus_tamriel_cyrodiil_leyawiin
			}
		}
        NOT = { #can only do it once
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:created_ojel_culture
			}
        }
    }

	is_valid = {
		highest_held_title_tier <= tier_kingdom #You don't want a subservient culture if you're a high king
        culture = { has_cultural_tradition = tradition_agaceph_core }
		capital_province = {
			OR = {
				geographical_region = mundus_tamriel_black_marsh_gideon
				geographical_region = mundus_tamriel_cyrodiil_leyawiin
			}
		}
		#Acceptance with Nibenese
		OR = {
			culture:nibenu = {
				cultural_acceptance = {
					target = root.culture
					value >= 60
				}
			}
			culture:nibenese = {
				cultural_acceptance = {
					target = root.culture
					value >= 60
				}
			}
		}
		#Need to be an imperial faith or liege needs to be an imperial
		OR = {
			custom_tooltip = {
				faith.religion = religion:imperial_religion
				text = ojel_follows_imperial_religion
			}
			
			custom_tooltip = {
				top_liege.culture = { has_cultural_tradition = tradition_imperial_legacy }
				text = top_liege_has_imperial_legacy
			}
		}
	}

	effect = { #EK TODO: event about picking nibenu or nibenese core
		culture:ojel = { 
			save_scope_as = ojel_culture
			culture_become_alive_effect = yes
			hidden_effect = {
				get_all_innovations_from = culture:agaceph
				reset_culture_creation_date = yes
			}
			 
		}
		set_culture = culture:ojel
		trigger_event = {
			id = south_tamriel.0014
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:created_ojel_culture
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
