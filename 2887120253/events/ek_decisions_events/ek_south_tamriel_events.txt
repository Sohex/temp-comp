scripted_trigger convert_random_ojel_counties_basic_reqs_trigger = {
	tier = tier_county
	NOT = { culture = culture:ojel }
}

scripted_effect convert_ojel_counties_effect = {
	# Several counties within your primary title will convert.
	capital_county = { set_county_culture = culture:ojel }
	while = {
		count = 3
		capital_province.duchy ?= {
			random_in_de_jure_hierarchy = {
				limit = { convert_random_ojel_counties_basic_reqs_trigger = yes }
				set_county_culture = culture:ojel
			}
		}
	}
}
