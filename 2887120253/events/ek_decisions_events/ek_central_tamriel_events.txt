# Colovia and Nibenay held by the same person
# Did Colovia inherit the other part, or the opposite?
# Was it by war, inheritance, or other?
central_tamriel.0002 = {
	type = character_event
	content_source = dlc_ek
	title = central_tamriel.0002.t

	desc = {
		first_valid = {
			triggered_desc = {
			    trigger = { has_character_flag = potentate_rejected_emperorship }
			    desc = central_tamriel.0002.desc_potentate
			}
			desc = central_tamriel.0002.desc
		}
	}

	theme = diplomacy_foreign_affairs_focus

	override_background = { reference = cyrodiil_white_gold }
	
	right_portrait = {
		character = root
		animation = personality_honorable
	}
	
	immediate = {		
		root = { save_scope_as = new_emperor }
	
		every_player = { 
			limit = { NOT = { this = root } }
			trigger_event = {
				days = 5 
				id = central_tamriel.0003
			}
		}
	}

	# imperial dynasty naming widget!
	widget = {
		gui = "event_window_widget_enter_text"
		container = "dynamic_birth_name"
		is_shown = {
			NOT = { has_character_flag = potentate_rejected_emperorship } # don't bother for Potentates who do not claim the empire
		}
		controller = {
			type = text
			data = {
				key = new_imperial_house_name
				default = {
					localization_key = blackdrake_imperial_house_name #reachmen get this option
					trigger = {
						culture = {
							OR = {
								this = culture:atmoran
								any_parent_culture_or_above = {
									this = culture:atmoran
								}
							}
						}
					}
				}
				default = {
					localization_key = lex_imperial_house_name #specific dynasty rename test
					trigger = {
						dynasty = dynasty:lex
					}
				}
				# default = {
				# 	localization_key = septim_imperial_house_name #for whenever that is a thing
				# 	trigger = {
				#		dynasty = dynasty:dyn_septim
				#	}
				# }
				default = {
					localization_key = old_imperial_house_name # by default, just use your current house name
				}
			}			
		}
		setup_scope = { root = { save_scope_as = text_target }}
	}

	# Adopt a new ruling name
	option = {
		name = central_tamriel.0002.a

		custom_tooltip = central_tamriel.0002.a.tooltip

		dynasty = {
			if = {
				limit = { root = { has_variable = new_imperial_house_name }}
				set_dynasty_name = new_imperial_house_name_default_key
			}
			else = {
				set_dynasty_name = new_imperial_house_name_entered_text
			}
			add_dynasty_modifier = {
				modifier = imperial_dynasty_modifier
			}
		}

		add_character_flag = emperor_renamed_dynasty

		ai_chance = 100 #have AI always pick this, as it'll usually just use their base name anyway but we wanna force the few special options there are

		# don't bother showing this for characters who reject emperorship
		trigger = {
			NOT = { has_character_flag = potentate_rejected_emperorship }
		}
	}

	# Keep your current name
	option = {
		name = central_tamriel.0002.b

		custom_tooltip = central_tamriel.0002.b.tooltip

		dynasty = {
			add_dynasty_modifier = {
				modifier = imperial_dynasty_modifier
			}
		}

		ai_chance = 0

		# don't bother showing this for characters who reject emperorship
		trigger = {
			NOT = { has_character_flag = potentate_rejected_emperorship }
		}
	}

	# Third option for Potentates
	option = {
		name = central_tamriel.0002.c

		dynasty = {
			add_dynasty_modifier = {
				modifier = potentate_dynasty_modifier
			}
		}

		ai_chance = 0

		# only show this for Potentates who rejected becoming emperor
		trigger = {
			has_character_flag = potentate_rejected_emperorship
		}
	}

	after = {
		create_artifact_red_dragon_crown_effect = { OWNER = root }
		if = {
			limit = {
				has_character_flag = emperor_renamed_dynasty
			}
			house = {
				set_house_name = new_imperial_house_name_entered_text_cheat # for whatever reason, set_house_name only accepts loc keys. so we assign this after assigning the dynasty rename
			}
			remove_character_flag = emperor_renamed_dynasty
		}

		if = {
			limit = { has_variable = new_imperial_house_name }
			remove_variable = new_imperial_house_name
		}
		# don't need this one anymore
		if = {
			limit = {
				has_character_flag = potentate_rejected_emperorship
			}
			remove_character_flag = potentate_rejected_emperorship
		}

	}
}

# Event to inform player of Cyrodiil being united
central_tamriel.0003 = {
	type = character_event
	content_source = dlc_ek
	title = central_tamriel.0003.t

	desc = {
		first_valid = {
			triggered_desc = {
			    trigger = { scope:new_emperor = { has_title = title:e_potentate }}
			    desc = central_tamriel.0003.desc_potentate
			}
			desc = central_tamriel.0003.desc
		}
	}
	
	theme = diplomacy_foreign_affairs_focus

	override_background = { reference = cyrodiil_white_gold }
	
	right_portrait = {
		character = scope:new_emperor
		animation = personality_honorable
	}

	option = {
		name = central_tamriel.0003.a
	}

	option = { #Join Empire
		name = emperor_zero.0045.c

		trigger = {
			highest_held_title_tier <= tier_kingdom
			is_independent_ruler = yes
			is_landed = yes
			OR = {
				OR = {
					title:e_cyrodiil = { is_de_jure_liege_or_above_target = root.primary_title } #De jure part of Cyrodiil
				}				
				culture = { #Imperial
					OR = {
						has_cultural_tradition = tradition_imperial_legacy
						has_cultural_tradition = tradition_cyrodiilic_subject
					}
				}
			}
			NOT = { government_has_flag = government_is_pirate } #These guys are not real rulers
		}

		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
		}
		change_liege = {
			liege = scope:tiber_character
			change = scope:change
		}
		add_opinion = {
			modifier = became_vassal
			target = scope:tiber_character
			opinion = 10
		}
		resolve_title_and_vassal_change = scope:change

		ai_chance = {
			base = 50
			modifier = {
				add = -50
				OR = {
					has_trait = ambitious
					has_trait = arrogant	
				}
			}
			modifier = {
				add = 50
				OR = {
					has_trait = content
					has_trait = craven
					has_trait = lazy
				}
			}
			modifier = {
				add = 50
				has_trait = zealous
				faith.religion = {
					OR = {
						this = religion:atmoran_religion
						this = religion:imperial_religion
					}
				}
			}
			modifier = {
				add = -50
				has_trait = zealous
				faith.religion = {
					NOR = {
						this = religion:atmoran_religion
						this = religion:imperial_religion
					}
				}
			}
			modifier = {
				add = 50
				OR = {
					has_trait = education_intrigue
					has_trait = education_diplomacy
					has_trait = education_stewardship
					has_trait = education_learning
				}
			}
		}
	}
}

# Dawn Court becomes a Cyrodilic Cult
central_tamriel.0009 = {
	type = character_event
	content_source = dlc_ek
	title = central_tamriel.0009.t
	desc = central_tamriel.0009.desc
	
	theme = faith

	override_background = { reference = cyrodiil_white_gold }

	
	immediate = {
		show_as_tooltip = {
			faith:court_dawnbeauty = {
				add_doctrine = special_doctrine_cyrodiilic_cult
			}
			# If the Council of the Eight has been formed, add Dawn Court to it
			if = {
				limit = {
					exists = title:k_council_of_the_eight.holder
				}
				# if the CotE is formed, grab its head
				title:k_council_of_the_eight.holder = {
					save_scope_as = cote_head
				}
				custom_tooltip = dawn_court_gain_spiritual_head
				custom_tooltip = form_cote_religious_head
				hidden_effect = {
					faith:court_dawnbeauty = {
						#remove all other HoF doctrines to be safe
						if = {
							limit = { has_doctrine = doctrine_no_head }
							remove_doctrine = doctrine_no_head
						}
						if = {
							limit = { has_doctrine = doctrine_temporal_head }
							remove_doctrine = doctrine_temporal_head
						}
						if = {
							limit = { has_doctrine = doctrine_mystical_birthright_head }
							remove_doctrine = doctrine_mystical_birthright_head
						}
						add_doctrine = doctrine_spiritual_head #then add the relevant one	
						if = {
							limit = { has_doctrine = doctrine_clerical_succession_spiritual_appointment }
							remove_doctrine = doctrine_clerical_succession_spiritual_appointment
						}
						if = {
							limit = { has_doctrine = doctrine_clerical_succession_temporal_appointment }
							remove_doctrine = doctrine_clerical_succession_temporal_appointment
						}
						if = {
							limit = { has_doctrine = doctrine_clerical_succession_temporal_fixed_appointment }
							remove_doctrine = doctrine_clerical_succession_temporal_fixed_appointment
						}
						add_doctrine = doctrine_clerical_succession_spiritual_fixed_appointment
						set_religious_head_title = title:k_council_of_the_eight #set the HoF to the CotE (gotta do this after giving them the appropriate doctrine)
					}
				}
			}
		}
	}
	
	left_portrait = {
		character = root
		animation = personality_zealous
	}

	lower_right_portrait = {
		character = scope:cote_head
	}
	
	option = {
		name = central_tamriel.0009.a
		flavor = central_tamriel.0009.a.flavor
	}
}
