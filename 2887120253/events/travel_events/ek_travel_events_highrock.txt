# The Witches of Hags Fen
ek_travel_events_highrock.4000 = {
	type = character_event
	content_source = dlc_ek
	title = ek_travel_events_highrock.4000.t
	desc = ek_travel_events_highrock.4000.desc
	theme = travel_danger
	override_background = ep2_hunt_foggy_forest
	override_effect_2d = { reference = fog }
	widget = { gui = event_window_widget_vfx_fog container = foreground_shader_vfx_container }
	left_portrait = {
		character = root 
		animation = paranoia
	}

	right_portrait = {
		character = scope:savior
		scripted_animation = duel_wield_weapon
		camera = camera_event_center
	}

	cooldown = { years = 2 }
	trigger = {
		is_available_travelling_adult = yes
		 
		is_location_valid_for_travel_event_on_land = yes
		location = {
			geographical_region = mundus_tamriel_high_rock_glenumbra
		}
		is_witch_trigger = no
		current_travel_plan ?= {
			any_entourage_character = {
				NOT = { this = root }
				count > 1
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			factor = 5
			has_variable = bad_prophecy_var
		}
	}

	immediate = {
		mp_delay_travel_plan = { DAYS = 90 }
		traveler_danger_xp_effect = {
			MIN = 3
			MAX = 7
		}
		remove_short_term_gold = minor_gold_value
		current_travel_plan = {
			ordered_entourage_character = {
				limit = { NOT = { this = root } }
				order_by = {
					value = 0
					add = prowess
					if = {
						limit = { is_knight_of = root }
						add = 50
					}
					if = {
						limit = {
							root = { exists = cp:councillor_marshal }
							this = root.cp:councillor_marshal
						}
						add = 50
					}
					if = {
						limit = { has_court_position = bodyguard_court_position }
						add = 50
					}
				}
				save_scope_as = savior
			}
		}
	}
	
	option = { # fight your way out
		name = ek_travel_events_highrock.4000.a
		duel = {
			skill = prowess
			value = high_skill_rating
			10 = {
				desc = ek_travel_events_highrock.4000.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1
				}
				send_interface_toast = {
					left_icon = root
					type = event_toast_effect_good
					title = tooltip.ek_hunt.1003.a.success
					add_prestige = medium_prestige_gain
				}
			}
			
			5 = {
				desc = ek_travel_events_highrock.4000.a.failure
				send_interface_toast = {
					left_icon = root
					type = event_toast_effect_bad
					title = tooltip.ek_hunt.1003.a.failure
					add_prestige = medium_prestige_loss
					random_list = {
						10 = {
							death = { death_reason = death_hag }
						}
						25 = { 
							increase_wounds_effect = { REASON = hag }
						} 
					} 
				}
			}
		}
	}

	option = { # become a witch
		name = ek_travel_events_highrock.4000.b
		trigger = {
			is_female = yes
		}
		give_witch_secret_or_trait_effect = yes
		current_travel_plan = {
			delay_travel_plan = { days = 7 }
		}
	}

	option = { 
		name = ek_travel_events_highrock.4000.c
		increase_wounds_effect = { REASON = hag }
	}
}

					left_icon = root
					type = event_toast_effect_good
					title = tooltip.ek_hunt.1003.a.success
					add_prestige = medium_prestige_gain
				}
			}
			
			5 = {
				desc = ek_travel_events_highrock.4005.a.failure
				send_interface_toast = {
					left_icon = root
					type = event_toast_effect_bad
					title = tooltip.ek_hunt.1003.a.failure
					add_prestige = medium_prestige_loss
					random_list = {
						10 = {
							death = { death_reason = death_harpy }
						}
						25 = { 
							increase_wounds_effect = { REASON = harpy }
						} 
					} 
				}
			}
		}

		stress_impact = {
			craven = major_stress_impact_gain
			brave = medium_stress_impact_loss	
		}

		ai_chance = {
			base = 0
			modifier = {
				has_trait = brave 
				add = 50
			}
			modifier = {
				prowess >= high_skill_rating
				add = 50
			}
		}
	}

	option = { # If mercenaries are hired, use them
		name = ek_travel_events_highrock.4005.b
		custom_tooltip = ek_travel_events_highrock.4005.b.t

		trigger = { current_travel_plan = { has_travel_option = hire_experienced_mercenaries_option } }

		stress_impact = {
			craven = minor_stress_impact_loss	
			brave = medium_stress_impact_gain	
		}

		ai_chance = {
			base = 100
		}
	}

	option = { # Use destruction magic
		name = ek_travel_events_highrock.4005.c
		reason = destruction

		trigger = { 
			has_perk = destruction_novice_perk
			has_magicka_more_than = { VALUE = t1_spell_cost }
		}

		show_as_unavailable = {
			has_magicka_less_than = { VALUE = t1_spell_cost }
		}

		add_magicka = { VALUE = t1_spell_cost.negate }

		ai_chance = {
			base = 100
		}
	}
	
	option = { # Run away!
		name = ek_travel_events_highrock.4005.d
		increase_wounds_effect = { REASON = harpy }

		stress_impact = {
			craven = minor_stress_impact_loss	
			brave = major_stress_impact_gain	
		}

		ai_chance = {
			base = 50
		}	
	}
}
