# The Hill of Kvatch
ek_travel_events_cyrodiil.0005 = {
	type = character_event
	content_source = dlc_ek
	title = ek_travel_events_cyrodiil.0005.t
	desc = ek_travel_events_cyrodiil.0005.desc
	theme = travel
	
	left_portrait = {
		character = root
		animation = happiness
	}

	right_portrait = {
		character = scope:new_friend
		animation = ecstasy
	}

	cooldown = { years = 10 }

	trigger = {
		is_available_travelling_adult = yes
		 
		is_location_valid_for_travel_event_on_land = yes
		NOT = { location.county.holder = root } #If I own it then it does not make sense to visit it.
		location = {
			geographical_region = mundus_tamriel_cyrodiil_kvatch
		}
	}

	immediate = {
		play_music_cue = mx_cue_travel_arrival
		current_travel_plan = {
			random_entourage_character = {
				limit = {
					can_set_relation_friend_trigger = { CHARACTER = root }
					is_available_travelling_ai_adult = yes
				}
				save_scope_as = new_friend
			}
		}
	}

	option = { # very cool
		name = ek_travel_events_cyrodiil.0005.a
		skill = diplomacy
		add_diplomacy_lifestyle_xp = medium_lifestyle_xp
		current_travel_plan = {
			delay_travel_plan = { days = 7 }
		}
		random = {
			chance = 30
			set_relation_friend = { reason = friend_common_interests target = scope:new_friend }
		}
	}

	option = { # just go
		name = ek_travel_events_cyrodiil.0005.b
	}
}

# Among the Nixad
ek_travel_events_cyrodiil.4000 = {
	type = character_event
	content_source = dlc_ek
	title = ek_travel_events_cyrodiil.4000.t
	desc = ek_travel_events_cyrodiil.4000.desc
	theme = travel_danger
	left_portrait = {
		character = root 
		animation = shock
	}

	right_portrait = {
		character = scope:caravan_master 
		animation = sadness
	}

	cooldown = { years = 10 }
	trigger = {
		is_available_travelling_adult = yes
		 
		is_location_valid_for_travel_event_on_land = yes
		location = {
			geographical_region = mundus_tamriel_cyrodiil_gold_coast
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			factor = 5
			has_variable = bad_prophecy_var
		}
	}

	immediate = {
		mp_delay_travel_plan = { DAYS = 90 }
		traveler_danger_xp_effect = {
			MIN = 3
			MAX = 7
		}
		remove_short_term_gold = minor_gold_value
		domicile ?= {
			change_provisions = medium_provisions_loss
		}
		if = {
			limit = { exists = current_travel_plan.travel_leader }
			current_travel_plan.travel_leader = { save_scope_as = caravan_master }
		}
	}
	
	option = { # Damned things!
		name = ek_travel_events_cyrodiil.4000.a

		stress_impact = {
			base = medium_stress_impact_gain
		}
	}	
}
