# Tribal Connections
ek_travel_events_blackmarsh.0030 = {
	type = character_event
	content_source = dlc_ek
	title = ek_travel_events_blackmarsh.0030.t
	desc = ek_travel_events_blackmarsh.0030.desc
	theme = travel
	
	left_portrait = {
		character = root
		animation = personality_content
	}

	right_portrait = {
		character = scope:tribal
		animation = personality_bold
	}

	cooldown = { years = 10 }

	trigger = {
		is_available_travelling_adult = yes
		 
		is_location_valid_for_travel_event_on_land = yes
		faith = { NOT = { has_doctrine = tenet_kuvastei } }
		root.location.faith = { has_doctrine = tenet_kuvastei }
	}

	immediate = {
		play_music_cue = mx_cue_travel_arrival
		create_character = {
			template = servant_character
			location = root.location
			culture = root.location.culture
			faith = root.location.faith
			save_scope_as = tribal
		}
	}

	option = { # hmmm... forgiving is good
		name = ek_travel_events_blackmarsh.0030.a
		add_piety = medium_piety_loss
		stress_impact = {
			base = medium_stress_impact_loss
		}
		current_travel_plan = {
			delay_travel_plan = { days = 7 }
		}
	}

	option = { # tribal joins
		name = ek_travel_events_blackmarsh.0030.b
		add_piety = medium_piety_loss
		add_courtier = scope:tribal
		root.current_travel_plan = {
			add_companion = scope:tribal
		}
		scope:tribal = {
			add_character_flag = {
				flag = recruited
				days = 5
			}
		}
		reverse_add_opinion = {
			modifier = recruited_me_opinion
			target = scope:tribal
			opinion = 20
		}
		stress_impact = {
			base = medium_stress_impact_loss
		}
	}

	option = { # no thanks
		name = ek_travel_events_blackmarsh.0030.c
	}

	after = {
		hidden_effect = {
			if = {
				limit = {
					is_ai = yes
					scope:tribal = {
						NOT = {
							is_courtier_of = root
						}
					}
				}
				scope:tribal = {
					if = {
						limit = {
							root = { is_ai = yes }
						}
						silent_disappearance_effect = yes
					}
				}
			}
		}
	}
}

# On Voriplasms
ek_travel_events_blackmarsh.4000 = {
	type = character_event
	content_source = dlc_ek
	title = ek_travel_events_blackmarsh.4000.t
	desc = ek_travel_events_blackmarsh.4000.desc
	theme = travel_danger
	
	right_portrait = {
		character = root
		scripted_animation = duel_wield_weapon
	}

	lower_left_portrait = {
		character = scope:attacked_character
	}

	cooldown = { years = 2 }
	
	trigger = {
		is_available_travelling_adult = yes
		 
		is_location_valid_for_travel_event_on_land = yes
		location = {
			geographical_region = mundus_tamriel_black_marsh
		}
		current_travel_plan ?= {
			any_entourage_character = {
				NOT = { this = root }
				count > 1
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			factor = 5
			has_variable = bad_prophecy_var
		}
	}
	
	immediate = {
		mp_delay_travel_plan = { DAYS = 90 }
		current_travel_plan = {
			current_location = { save_scope_as = current_location }
			ordered_entourage_character = {
				limit = { NOT = { this = root } }
				order_by = {
					value = 0
					subtract = prowess
					if = {
						limit = { is_knight_of = root }
						add = -50
					}
					if = {
						limit = {
							root = { exists = cp:councillor_marshal }
							this = root.cp:councillor_marshal
						}
						add = -50
					}
					if = {
						limit = { has_court_position = bodyguard_court_position }
						add = -50
					}
				}
				save_scope_as = attacked_character
			}
		}
		traveler_danger_xp_effect = {
			MIN = 3
			MAX = 7
		}
	}

	option = { # Kill it! Kill it with fire!
		name = ek_hunt.1004.a
		duel = {
			skill = prowess
			value = high_skill_rating
			10 = {
				desc = ek_travel_events_blackmarsh.4000.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1
				}
				send_interface_toast = {
					left_icon = root
					type = event_toast_effect_good
					title = tooltip.ek_hunt.1003.a.success
					add_prestige = medium_prestige_gain
				}
			}
			
			5 = {
				desc = ek_travel_events_blackmarsh.4000.a.failure
				send_interface_toast = {
					left_icon = root
					type = event_toast_effect_bad
					title = ek_travel_events_blackmarsh.4000.a.failure
					add_prestige = medium_prestige_loss
					random_list = {
						10 = {
							death = { death_reason = death_voriplasm }
							scope:attacked_character = {
								death = { death_reason = death_voriplasm }
							}
						}
						25 = { 
							increase_wounds_effect = { REASON = voriplasm }
							current_travel_plan = {
								delay_travel_plan = { days = 7 }
							}
							scope:attacked_character = {
								death = { death_reason = death_voriplasm }
							}
						} 
					} 
				}
			}
		}

		stress_impact = {
			craven = minor_stress_impact_gain
		}
		
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_energy = 0.5
			}
		}
	}

	option = { # Run as fast as you can
		name = ek_hunt.1003.b
		custom_tooltip = ek_hunt.1002.b.tt
		stress_impact = {
			brave = minor_stress_impact_gain
		}

		scope:attacked_character = {
			death = { death_reason = death_voriplasm }
		}
		
		ai_chance = {
			base = 10
			modifier = {
				add = -10
				OR = {
					has_trait = irritable
					has_trait = brave
				}
			}
			modifier = {
				add = 10
				has_trait = craven
			}
		}
	}
}
