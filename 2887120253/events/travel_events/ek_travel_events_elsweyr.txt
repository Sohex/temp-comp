# Not Just Mounts
ek_travel_events_elsweyr.0001 = {
	type = character_event
	content_source = dlc_ek
	title = ek_travel_events_elsweyr.0001.t
	desc = ek_travel_events_elsweyr.0001.desc
	theme = travel
	left_portrait = {
		character = root 
		animation = shock
	}

	right_portrait = {
		character = scope:offended
		animation = anger
	}
	cooldown = { years = 10 }
	trigger = {
		is_available_travelling_adult = yes
		 
		is_location_valid_for_travel_event_on_land = yes
		NOT = { 
			culture = { has_cultural_pillar = heritage_khajiiti }
		}
		location = {
			culture = { has_cultural_pillar = heritage_khajiiti }
		}
	}
	
	immediate = {
		create_character = {
			template = servant_character
			location = root.location
			culture = root.location.culture
			faith = root.location.faith
			save_scope_as = offended
		}
	}

	option = { # Apologise
		name = ek_travel_events_elsweyr.0001.a
		add_prestige = medium_prestige_loss
		
		stress_impact = {   
			arrogant = miniscule_stress_impact_gain
			callous = miniscule_stress_impact_gain
			sadistic = miniscule_stress_impact_gain
		}
	}

	option = { # Give them an apology gift
		name = ek_travel_events_elsweyr.0001.b
		remove_short_term_gold = minor_gold_value
		
		stress_impact = {   
			arrogant = miniscule_stress_impact_gain
			callous = miniscule_stress_impact_gain
			sadistic = miniscule_stress_impact_gain
		}
	}

	option = { # so what
		name = ek_travel_events_elsweyr.0001.c
		add_prestige = minor_prestige_gain
		
		stress_impact = {   
			compassionate = major_stress_impact_gain
			craven = medium_stress_impact_gain
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = {
					is_ai = yes
					scope:offended = {
						NOT = {
							is_courtier_of = root
						}
					}
				}
				scope:offended = {
					if = {
						limit = {
							root = { is_ai = yes }
						}
						silent_disappearance_effect = yes
					}
				}
			}
		}
	}
}

	}
