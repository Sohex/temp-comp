### Attrebus assumes control of Cyrodiil, to deal with the fallout of the war
# Part 1: The Potentate
ek_attrebus_potentate.0001 = {
	type = character_event
	content_source = dlc_ek
	title = ek_attrebus_potentate.0001.t
	
	desc = ek_attrebus_potentate.0001.desc
	
	theme = war
	
	left_portrait = {
		character = scope:attrebus
		animation = war_over_win
	}
	
	right_portrait = {
		character = scope:potentate
		animation = prisondungeon
	}
	
	immediate = {
		scope:defender = { save_scope_as = potentate }
		scope:attacker = { save_scope_as = attrebus }

		#This is to prevent the game over event from allowing potentate players to become LAAMPs while imprisoned
		if = {
			limit = {
				scope:attrebus ?= { is_ai = yes }
			}
			add_character_flag = immediate_execution
		}
		
		# Setup the deal with Liodes - Give them all the counties within any duchy they hold land in
		every_in_list = {
			list = zenas_ally
			
			save_scope_as = current_zenas_ally
			
			hidden_effect = {
				every_realm_de_jure_duchy = {
					every_in_de_jure_hierarchy = {
						limit = {
							tier = tier_county
							holder.top_liege = scope:potentate
						}
						add_to_list = zenas_ally_give_land
					}
				}

				create_title_and_vassal_change = {
					type = conquest
					save_scope_as = change
					add_claim_on_loss = yes
				}
				
				every_in_list = {
					list = zenas_ally_give_land
					change_title_holder = {
						holder = scope:current_zenas_ally
						change = scope:change
						take_baronies = yes
					}
				}
				scope:current_zenas_ally = {
					change_liege = {
						liege = scope:attrebus
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
			
			custom_tooltip = attrebus_potentate_cb_victory_zenas_allies_deal_ct
			
			clear_variable_list = zenas_ally_give_land
		}
		
		## There are some stuff to take care of before usurping the titles, so we do that here and not in the cb
		# We save all the vassals of the former Potentate
		scope:potentate = {
			every_vassal_or_below = { add_to_list = potentate_vassals }
		}
		
		## Now we can usurp stuff
		hidden_effect = {
			destroy_title = title:c_nf_chorak
			scope:defender = { every_held_title = { add_to_temporary_list = titles_taken } }
			scope:attrebus = {
				if = { # Do we become administrative?
					limit = {
						scope:potentate = { government_has_flag = government_is_administrative }
						has_dlc_feature = roads_to_power
					}
					#save_scope_as = administrative_liege
					change_to_administrative_effect = yes
				}
			}
			
			create_title_and_vassal_change = {
				type = conquest_claim
				save_scope_as = change
				add_claim_on_loss = yes
			}
			
			scope:potentate = {
				if = {
					limit = { this = title:e_potentate.holder }
					title:e_potentate = {
						change_title_holder = {
							holder = scope:attrebus
							change = scope:change
						}
					}
				}
				every_vassal = {
					limit = {
						primary_title.tier < scope:attrebus.primary_title.tier
						NOT = {
							primary_title.tier = tier_barony
						}
					}
					change_liege = {
						liege = scope:attrebus
						change = scope:change
					}
				}
				change_liege = {
					liege = scope:attrebus
					change = scope:change
				}
				every_held_title = {
					limit = {
						tier > tier_barony
						NOT = { is_noble_family_title = yes }
					}
					change_title_holder = {
						holder = scope:attrebus
						change = scope:change
						take_baronies = yes
					}
				}
			}
			scope:potentate = {
				# Probably not necessary since c_nf_chorak is manually destroyed further up, but maybe this is needed
				# as a fallback for situations where the potentate somehow ends up someone else?
				every_held_title = {
					limit = {
						is_noble_family_title = yes
					}
					save_scope_as = title_to_destroy
				}
				if = {
					limit = {
						exists = scope:title_to_destroy
					}
					destroy_title = scope:title_to_destroy
				}
			}
			resolve_title_and_vassal_change = scope:change
			destroy_title = title:e_potentate
		}
		# We set up the warlord empire
		scope:attrebus = {
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
			}
			title:e_sosildor_empire = {
				change_title_holder = {
					holder = scope:attrebus
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			destroy_title = title:e_colovian_estates
			add_character_modifier = {
				modifier = warlord_emperor_modifier
			}
			
			hidden_effect = {
				title:e_sosildor_empire = {
					set_variable = {
						name = warlord_empire
						value = yes
					}
				}
			}

			# since the empire is titular, make it primo to not wack up succession
			if = {
				limit = {
					NOT = {
						has_realm_law = single_heir_succession_law
					}
				}
				add_realm_law_skip_effects = single_heir_succession_law
			}
		}
		# Theyre all traitors!
		dynasty:dyn_chorak = {
			every_dynasty_member = {
				hidden_effect = {
					scope:attrebus = {
						add_opinion = {
							target = prev
							modifier = treachery_opinion
							opinion = -100
						}
					}
				}
				if = {
					limit = {
						OR = {
							this = character:chorak_101
							top_liege ?= scope:attrebus
						}
						NOT = { is_landless_adventurer = yes }
						is_imprisoned = no
					}
					scope:attrebus = {
						imprison = {
							target = prev
							type = dungeon
						}
					}
				}
			}
		}

		#This is to prevent the game over event from allowing potentate players to become LAAMPs while imprisoned
		if = {
			limit = {
				has_character_flag = immediate_execution
			}
			execute_prisoner_effect = {
				VICTIM = scope:potentate
				EXECUTIONER = scope:attrebus
			}
			remove_character_flag = immediate_execution
		}
	}
	
	option = {
		name = ek_attrebus_potentate.0001.a
		add_prestige = 500
		add_piety = 250
		custom_tooltip = become_pretender_emperor_decision_title_break_attrebus #slightly different warning due to easier title scope
	}
	
	after = {
		every_player = {
			limit = {
				NOR = {
					this = scope:attrebus
					this = scope:potentate
					AND = {
						this = character:chorak_101b
						is_landless_adventurer = yes
					}
				}
			}
			trigger_event = {
				id = ek_attrebus_potentate.0002
				days = 1
			}
		}
		# Cazuras LAAMP gets alerted
		if = {
			limit = {
				character:chorak_103 = {
					is_alive = yes
					is_landless_adventurer = yes
				}
			}
			character:chorak_103 = {
				trigger_event = {
					id = ek_attrebus_potentate.0314
					days = { 7 21 }
				}
			}
		}
		#Rim-men stuff
		title:c_hakoshae.holder = {
			trigger_event = {
				id = ek_rimmen.0001
				days = { 7 21 }
			}
		}
		title:c_rimmen.holder = {
			trigger_event = {
				id = ek_rimmen.0001
				days = { 7 21 }
			}
		}
	}
}

			every_vassal = {
				limit = {
					culture = { has_cultural_tradition = tradition_colovian_core }
				}
				add_opinion = {
					target = root
					modifier = betrayal_opinion
					opinion = -25
				}
			}
			every_tributary = {
				limit = {
					culture = { has_cultural_tradition = tradition_colovian_core }
				}
				add_opinion = {
					target = root
					modifier = betrayal_opinion
					opinion = -25
				}
			}
		}
	}

	option = { # Go with Attrebus loyalist
		name = ek_attrebus_potentate.0064.c

		scope:imperial_legion_attrebus_loyalist = { save_scope_as = new_leader_first_legion }

	type = character_event

	}

	}
	
	immediate = {
		hidden_effect = {
			scope:potentate = {
				destroy_title = title:e_colovian_estates
				if = {
					limit = {
						scope:attrebus = { is_imprisoned = no }
					}
					imprison = {
						target = scope:attrebus
						type = dungeon
					}
				}
			}
			
			scope:attrebus = {
				#Attrebus loses all non-de Jure vassals of Chorrol
				primary_title = { save_scope_as = primary_title }
				every_vassal = {
					limit = { NOT = { this.primary_title = { target_is_de_jure_liege_or_above = scope:primary_title } } }
					create_title_and_vassal_change = {
						type = independency
						save_scope_as = change
						add_claim_on_loss = no
					}
					becomes_independent = {
						change = scope:change
					}

					resolve_title_and_vassal_change = scope:change
				}
				every_tributary = {
					custom = custom.every_defender_tributary
					end_tributary = yes #TODO_CD_MPO some sort of message here
				}
			}
			
			scope:attrebus = { trigger_event = ek_attrebus_potentate.0102 }
		}
		custom_tooltip = attrebus_potentate_potentate_win_colovian_estates_shattered
	}

	option = {
		name = ek_attrebus_potentate.0101.a
	}
	
	after = {
		every_player = {
			limit = { NOR = { this = scope:attrebus this = scope:potentate } }
			trigger_event = {
				id = ek_attrebus_potentate.0103
				days = 1
			}
		}
	}
}

ek_attrebus_potentate.0201 = {
	type = character_event
	content_source = dlc_ek
	hidden = yes
	
	trigger = {
		has_title = title:e_colovian_estates
		# We're in the war against the Potentate
		any_character_war = {
			using_cb = attrebus_potentate_cb
		}
	}
	
	immediate = {
		save_scope_as = attrebus
		
		random_character_war = {
			limit = { using_cb = attrebus_potentate_cb }
			this.primary_defender = { save_scope_as = potentate }
			
			end_war = invalidated
		}
		
		# The Colovian Estates title is destroyed
		destroy_title = title:e_colovian_estates
		
		# You lose all non-de Jure vassals of Chorrol
		primary_title = { save_scope_as = primary_title }
		every_vassal = {
			limit = { NOT = { this.primary_title = { target_is_de_jure_liege_or_above = scope:primary_title } } }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = change
				add_claim_on_loss = yes
			}
			becomes_independent = {
				change = scope:change
			}

			resolve_title_and_vassal_change = scope:change
		}

		every_tributary = {
			custom = custom.every_defender_tributary
			end_tributary = yes #TODO_CD_MPO some sort of message here
		}
		
		primary_heir = {
			# To ensure that if the player plays Attrebus they get this event
			trigger_event = {
				id = ek_attrebus_potentate.0202
				days = 3
			}
		}
		every_player = {
			# To ensure that if the player plays Attrebus they don't get the second event
			limit = { NOT = { has_primary_title = scope:primary_title } }
			trigger_event = {
				id = ek_attrebus_potentate.0203
				days = 3
			}
		}
	}
}
