emperor_zero.1004 = { #Emperor Zero dies too soon, pick heir
	type = character_event
	title = emperor_zero.1004.t
	desc =  emperor_zero.1004.desc
	theme = death
	
	right_portrait = {
		character = scope:emperor_zero_heir
		animation = personality_content
	}

	left_portrait = {
		character = scope:talos_character
		animation = personality_rational
	}

	immediate = {
		root = { save_scope_as = emperor_zero_heir }
	}

	option = {
		name = emperor_zero.1004.a
		trigger = {
			exists = scope:emperor_zero_heir
		}
		scope:emperor_zero_heir = {
			add_trait = emperor_zero
			add_character_flag = {
				flag = immune_to_disease
				years = 15
			}
		}
		scope:story = { make_story_owner = scope:emperor_zero_heir }
	}

	option = {
		name = emperor_zero.1004.b
		create_title_and_vassal_change = {
			type = returned
			save_scope_as = change
			add_claim_on_loss = no
		}
		root.capital_county = {
			change_title_holder = {
				holder = scope:talos_character
				change =scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		set_designated_heir_unsafe = scope:talos_character
		scope:talos_character = {
			add_trait = emperor_zero
			trigger_event = {
				id = emperor_zero.1006
				days = 1
			}
		}
		if = {
			limit = { is_ai = no }
			set_player_character = scope:talos_character
		}
		depose = yes
		scope:story = { make_story_owner = scope:talos_character }
		ai_chance = 100
	}
}

#Tiber comes to your court
emperor_zero.0031 = {
	type = character_event
	title = emperor_zero.0031.t
	desc =  emperor_zero.0031.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = scope:talos_character
		animation = personality_rational
	}

	immediate = {
		root = { save_scope_as = emperor_zero_character }
		if = {
			limit = { has_game_rule = random_talos_culture }
			create_character = {
				#name = "Hjalti" - random name
				location = scope:emperor_zero_character.location
				template = random_talos_character_template
				save_scope_as = talos_character
			}
		}
		else_if = {
			create_character = {
				name = "Hjalti"
				location = scope:emperor_zero_character.location
				template = talos_character_template
				save_scope_as = talos_character
			}
		}

		if = {
			limit = { scope:talos_character.culture ?= culture:northmen }
			scope:talos_character = {
				random_list = {
					50 = {
						create_dynasty = { name = "dynn_erle_baird" }
						house = { set_house_name = dynn_erle_baird }
					}
					50 = {
						if = {
							limit = { is_female = no }
							give_nickname = nick_early_beard 
						}
						else = {
							give_nickname = nick_early_braid
						}
					}
				}
			}
		}
		if = {
			limit = { scope:talos_character.culture ?= { has_cultural_pillar = heritage_atmoran } }
			scope:talos_character = {
				random_list = {
					50 = {
						create_dynasty = { name = "dynn_Early_Beard" }
						house = { set_house_name = dynn_Early_Beard }
					}
					50 = {
						if = {
							limit = { is_female = no }
							give_nickname = nick_early_beard 
						}
						else = {
							give_nickname = nick_early_braid
						}
					}
				}
			}
		}
	}


	option = { #Accept Talos
		name = emperor_zero.0031.a
		add_courtier = scope:talos_character
		random_owned_story = {
			type = story_emperor_zero
			set_variable = {
				name = talos_general
				value = scope:talos_character
			}
		}
		hidden_effect = {
			scope:talos_character = { #This is needed to not have Talos's liege marry him off
				set_variable = can_not_marry
			}
		}
		ai_chance =  {
			base = 100
			modifier = {
				has_trait = arrogant
				add = -100
			}
		}
	}

	option = { #Reject Talos
		name = emperor_zero.0031.b
		ai_chance =  {
			base = 0
			modifier = {
				has_trait = arrogant
				add = 100
			}
		}
		scope:talos_character = {
			create_landless_adventurer_title_effect = {
				REASON = flag:exile
				FLAVOR_CHAR = scope:emperor_zero_character
			}
			set_relation_nemesis = { reason = nemesis_historical target = scope:emperor_zero_character }
		}
	}

	after = {
		scope:talos_character = {
			custom_tooltip = hjalti_change_name_tooltip
			give_nickname = nick_stormcrown
		}
		hidden_effect = {
			scope:talos_character = { 
				change_first_name = "Talos"
				set_culture = culture:hrothgar
				add_character_flag = {
					flag = immune_to_disease
					years = 15
				}
				add_character_flag = {
					flag = immune_to_harm_events
				}
				add_character_flag = {
					flag = immune_to_elderly_health_flag
				}
				add_prestige = 5000
				if = {
					limit = {
						is_lowborn = yes
					}
					create_dynasty = {
						name = "dynn_of_Atmora"
						coat_of_arms = dyn_atmora
					}
					house = { set_house_name = dynn_of_Atmora }
				}
				else = {
					dynasty = { set_dynasty_name = dynn_of_Atmora }
					house = { set_house_name = dynn_of_Atmora }
				}
			}
		}
		if = {
			limit = { scope:talos_character = { is_landless_adventurer = yes } }
			if = { 
				limit = { scope:talos_character = { is_female = yes } }
				scope:talos_character.primary_title = { set_title_name = "DAUGHTERS_OF_ATMORA" }
			}
			else = {
				scope:talos_character.primary_title = { set_title_name = "SONS_OF_ATMORA" }
			}	
		}
	}
}

#Cuhlecain becomes emperor, Hjalti tries to kill him
emperor_zero.0041 = {
	type = character_event
	title = emperor_zero.0041.t
	desc =  emperor_zero.0041.desc
	theme = intrigue
	override_background = {
		reference = cyrodiil_white_gold
	}

	immediate = {
		root = { save_scope_as = emperor_zero }
		random_owned_story = {
			type = story_emperor_zero
			var:talos_general = { save_scope_as = talos_character }
		}
		set_global_variable = {
			name = emperor_zero_assassination_attempt
			value = yes
		}
		if = {
			limit = { 
				exists = scope:emperor_zero.primary_heir
				scope:emperor_zero.primary_heir != scope:talos_character
			}
		}
	}

	left_portrait = {
		character = root
		animation = fear
	}

	right_portrait = {
		character = scope:talos_character
		animation = aggressive_dagger
	}

	option = { #Surrender to Talos
		name = emperor_zero.0041.a
		create_title_and_vassal_change = {
			type = returned
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:c_imperial_city = {
			change_title_holder = {
				holder = scope:talos_character
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:emperor_zero = {
			set_designated_heir_unsafe = scope:talos_character
			add_character_flag = killed_by_hjalti_flag
		}
		if = {
			limit = { is_ai = no }
			set_player_character = scope:talos_character
		}
		ai_chance = 100
	}

	option = { #Resist
		name = emperor_zero.0041.b

		duel = {
			skill = prowess
			target = scope:talos_character

			30 = {
				desc = emperor_zero.0041.b.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2
					min = -50
				}
				trigger_event = emperor_zero.0043
			}
			70 = {
				desc = emperor_zero.0041.b.fail
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2
					min = 50
				}
				trigger_event = emperor_zero.0044
			}
		}
	}

	after = {
		if = {
			limit = { scope:emperor_zero = { has_character_flag = killed_by_hjalti_flag } }
			scope:emperor_zero = { death = { death_reason = death_murder } }
			scope:talos_character = {
				trigger_event = {
					id = emperor_zero.0042
					days = 1
				}
			}
		}
	}
}
